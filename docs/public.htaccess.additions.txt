# Add to the TOP of public/.htaccess (keep the stock Laravel rewrite block below it).
# Blocks dotfiles (.env, .git ...) and common backup/source files if the document root is ever misconfigured.
<FilesMatch "^\.(?!well-known)">
    Require all denied
</FilesMatch>
<FilesMatch "(?i)\.(env|bak|old|orig|sql|log|ini|sh|lock|md)$">
    Require all denied
</FilesMatch>
Options -Indexes
